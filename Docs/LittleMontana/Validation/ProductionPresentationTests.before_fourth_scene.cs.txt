        }
        [Test] public void EveryRoleAndRequiredIconHasAnImportedNonemptySprite()
        {
            var art=AssetDatabase.LoadAssetAtPath<PresentationArt>(ProductionArtImporter.ArtPath);Assert.That(art,Is.Not.Null);Assert.That(art.sprites.Length,Is.EqualTo(70));
            Assert.That(art.sprites.Select(x=>x.id).Distinct().Count(),Is.EqualTo(70));
            foreach(var entry in art.sprites){Assert.That(entry.sprite,Is.Not.Null,entry.id);Assert.That(entry.sprite.rect.width,Is.GreaterThan(8));}
            var defs=AssetDatabase.LoadAssetAtPath<GameContent>(GameScaffold.ContentPath).Load();

            Assert.Throws<InvalidOperationException>(()=>art.Get("unmapped_required_icon"));
        }
        [Test] public void AuthoredWorldUsesAllCharactersRoutesAndOriginalAudioWithoutChangingContent()
        {
            var setup=EditorSceneManager.GetSceneManagerSetup();
            try
            {
                EditorSceneManager.OpenScene(GameScaffold.GameScene);var root=UnityEngine.Object.FindFirstObjectByType<GameRoot>();var p=root.Presentation;
                Assert.That(p,Is.Not.Null);Assert.That(p.characters.Length,Is.EqualTo(8));Assert.That(root.Sites.Length,Is.EqualTo(56));
                var defs=AssetDatabase.LoadAssetAtPath<GameContent>(GameScaffold.ContentPath).Load();Assert.That(p.routes.Length,Is.EqualTo(defs.routes.Count(r=>defs.instances.Single(i=>i.id==r.from).cityId=="burning_hills")));
                var plan=JsonUtility.FromJson<BurningHillsLayoutInstaller.LayoutPlan>(File.ReadAllText(BurningHillsLayoutInstaller.Folder+"/BurningHillsLayout.json"));
                foreach(var route in p.routes)
                {
                    var r=plan.routes.Single(x=>x.id==route.routeId);
                    Assert.That(Vector2.Distance(DistrictMap.ToPlan(route.Position(0)),r.points.First()),Is.LessThan(.01));
                    Assert.That(Vector2.Distance(DistrictMap.ToPlan(route.Position(1)),r.points.Last()),Is.LessThan(.01));
                    Assert.That(route.points.Length,Is.EqualTo(r.points.Length));
                }
                Assert.That(p.ambience.clip.samples,Is.EqualTo(2646000));Assert.That(p.ambience.loop,Is.True);Assert.That(p.eventAudio,Is.Not.Null);
                Assert.That(EditorBuildSettings.scenes.Where(x=>x.enabled).Select(x=>x.path),Is.EquivalentTo(new[]{GameScaffold.BootstrapScene,GameScaffold.GameScene,BigRockCityAuthoring.ScenePath,SevenHillsAuthoring.ScenePath}));
                Assert.That(p.explicitBuildingApproval,Is.True);Assert.That(p.releaseRequested,Is.False);
            }
            finally{if(setup.Length>0)EditorSceneManager.RestoreSceneManagerSetup(setup);}
        }
        [Test] public void ProductionReauthoringPreservesManualSceneAndPrefabBytes()
        {
            var paths=Directory.GetFiles(ProductionArtImporter.Root,"*",SearchOption.AllDirectories).Concat(new[]{GameScaffold.GameScene,GameScaffold.Prefabs+"/BuildingSite.prefab",GameScaffold.Prefabs+"/ListRow.prefab"}).ToArray();
            var before=paths.ToDictionary(x=>x,FileDigest);ProductionFinishInstaller.Apply();
            foreach(var pair in before)Assert.That(FileDigest(pair.Key),Is.EqualTo(pair.Value),pair.Key);
        }
        private static string FileDigest(string path)
        {
            using(var hash=SHA256.Create())using(var stream=File.OpenRead(path))return Convert.ToBase64String(hash.ComputeHash(stream));
        }

        [Test] public void OldOrUnboundAcceptanceCannotOpenTheReleaseGate()
        {
            Assert.That(ProductionVisualPreflight.AcceptsReceipt("{\"passed\":true}","current"),Is.False);
            Assert.That(ProductionVisualPreflight.AcceptsReceipt("{\"passed\":true,\"fingerprint\":\"old\"}","current"),Is.False);
            Assert.That(ProductionVisualPreflight.AcceptsReceipt("{\"passed\":false,\"fingerprint\":\"current\"}","current"),Is.False);
            Assert.That(ProductionVisualPreflight.AcceptsReceipt("{\"passed\":true,\"fingerprint\":\"current\"}","current"),Is.True);
        }
